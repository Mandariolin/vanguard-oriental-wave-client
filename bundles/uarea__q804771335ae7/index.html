var u_area_menu=!1;$("#uarea_nav, #_lmenu_overlay").on("click",function(){u_area_menu?($("#uarea_nav").removeClass("menu_open"),$("#user_navigator").removeClass("menu_open"),$("#_lmenu_overlay").stop(!0,!1).fadeOut(400),u_area_menu=!1):($("#uarea_nav").addClass("menu_open"),$("#user_navigator").addClass("menu_open"),$("#_lmenu_overlay").stop(!0,!1).fadeIn(400),u_area_menu=!0)})
